    \def\boxhead#1{{\bfseries #1}\par\nointerlineskip\vspace{3pt}%

    \begin{tikzpicture}[
        every node/.style={font=\sffamily\scriptsize},
        box/.style={code, text width=4.6cm, inner sep=4pt, align=left, font=\sffamily\scriptsize,
                    execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
        flow/.style={->, thick},
        note/.style={font=\sffamily\scriptsize},
    ]
        % skewer axis S = centre of the 5.0cm prompt box; right column starts at R
        \coordinate (S) at (2.64,0);
        \coordinate (R) at (6.78,0);

        % Left skewer: example.e, prompt, AutoProof, example.e (edited) ---------------------------------------------
        \node (init) [box, anchor=north] at (S) {\boxhead{example.e}
\begin{lstlisting}[xleftmargin=0.1cm]

feature
  min (x, y: INTEGER): INTEGER

      Result := x
    ensure

    end

        };

        \coordinate (row1) at ([yshift=-14mm]init.south);

        % Left column: prompt --------------------------------------------------------
        \node (prompt) [box, text width=5.0cm, anchor=north] at (row1 -| S) {\boxhead{Prompt}
\llmrole{System:} You are a coding assistant, expert in the Eiffel programming
language and AutoProof static verifier. [\dots\unkern] You must ONLY modify the
feature body [\dots\unkern]\par\smallskip
\llmrole{User:} The following feature does not verify. Please rewrite it so
that it verifies. [\dots\unkern]\par\smallskip
Context: Identifiers available in the post-state for the postcondition:
Current: EXAMPLE, x: INTEGER, y: INTEGER, Result: INTEGER. [\dots\unkern]\par\smallskip
Feature to fix:\par\smallskip
\begin{lstlisting}[xleftmargin=0.1cm]
min (x, y: INTEGER): INTEGER
    -- Smaller of x and y.
  do
    Result := x
  ensure
    res: (Result = x or Result = y)
      and Result <= x and Result <= y
  end
\end{lstlisting}
\smallskip
AutoProof error: Postcondition res may be violated. [\dots\unkern]
        };

        % Right column: LLM response, subclass -----------------------------------------
        \node (cand) [box, anchor=north west] at (row1 -| R) {\boxhead{LLM response}
Here is the corrected feature:\par\smallskip
\begin{lstlisting}[xleftmargin=0.1cm]
min (x, y: INTEGER): INTEGER
    -- Smaller of x and y.
  do
    if x < y then
      Result := x
    else
      Result := y
    end
  ensure
    res: (Result = x or Result = y)
      and Result <= x and Result <= y
  end
\end{lstlisting}
\smallskip
The conditional assigns the smaller argument to Result, so the postcondition holds.
        };

        \node (sub) [box, anchor=north west] at ([yshift=-12mm]cand.south -| R)
            {\boxhead{llm\_instrumented\_example.e}
\begin{lstlisting}[xleftmargin=0.1cm]
class LLM_INSTRUMENTED_EXAMPLE
inherit EXAMPLE redefine min end
feature
  min (x, y: INTEGER): INTEGER
    do
      if x < y then
        Result := x
      else
        Result := y
      end
    end
end
\end{lstlisting}
        };

        \coordinate (vy) at ([yshift=-6mm]prompt.south);

    end
end
\end{lstlisting}
        };

        \draw[flow] (verif.south) -- (verif.south |- after.north)
            node[midway, right, note, align=left]
                {succeeds: copy body into\\the original feature (LSP edit)};
